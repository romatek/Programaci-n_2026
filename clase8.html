8. El Patrón de Validación con Funciones

Al unir funciones con un bucle while, el programa principal queda limpio y se lee como español:

function limpiarPatente(texto) {
    return texto.trim().toUpperCase();
}

function validarPatente(patente) {
    return patente.length === 7;
}

let patente = limpiarPatente(prompt("Patente:"));
while (!validarPatente(patente)) {
    patente = limpiarPatente(prompt("Inválida. Ingrese 7 caracteres:"));
}
